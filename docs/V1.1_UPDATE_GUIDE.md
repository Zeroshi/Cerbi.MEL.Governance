### 3. Update MEL Options

**File:** `Cerbi.MEL.Governance/CerbiMelOptions.cs`

```csharp
using Cerbi.MEL.Governance.Services;

namespace Cerbi.MEL.Governance;

public class CerbiMelOptions
{
    // Identity
    private string? _explicitTenantId;
    private string? _explicitEnvironment;
    private GovernanceConfigReader? _configReader;

    public string? AppName { get; private set; }
    public string? ServiceName { get; private set; }
    public string? GovernanceProfile { get; private set; }

    // Queue settings
    public string? QueueConnectionString { get; private set; }
    public string? QueueName { get; private set; }

    // Fluent methods
    public CerbiMelOptions WithTenantId(string tenantId)
    {
        _explicitTenantId = tenantId;
        return this;
    }

    public CerbiMelOptions WithAppName(string appName)
    {
        AppName = appName;
        return this;
    }

    public CerbiMelOptions WithServiceName(string serviceName)
    {
        ServiceName = serviceName;
        return this;
    }

    public CerbiMelOptions WithGovernanceConfig(string? configPath = null)
    {
        _configReader = new GovernanceConfigReader(configPath);
        return this;
    }

    public CerbiMelOptions WithEnvironment(string environment)
    {
        _explicitEnvironment = environment;
        return this;
    }

    public CerbiMelOptions WithQueue(string connectionString, string queueName)
    {
        QueueConnectionString = connectionString;
        QueueName = queueName;
        return this;
    }

    // Resolution methods
    public string ResolveTenantId()
    {
        if (!string.IsNullOrEmpty(_explicitTenantId))
            return _explicitTenantId;

        var configTenant = _configReader?.GetTenantId();
        if (!string.IsNullOrEmpty(configTenant))
            return configTenant;

        var envTenant = Environment.GetEnvironmentVariable("CERBI_TENANT_ID");
        if (!string.IsNullOrEmpty(envTenant))
            return envTenant;

        throw new InvalidOperationException(
            "TenantId not configured. Use .WithGovernanceConfig() or .WithTenantId()");
    }

    public string ResolveEnvironment() => _explicitEnvironment ?? EnvironmentDetector.Environment;
    public string ResolveInstanceId() => EnvironmentDetector.InstanceId;
    public string? ResolveAppVersion() => EnvironmentDetector.AppVersion;
    public string? ResolveDeploymentId() => EnvironmentDetector.DeploymentId;
}
```

### 4. Update Logger

Update the logger to use resolved values and send ScoringEventDto:

**File:** `Cerbi.MEL.Governance/CerbiMelLogger.cs`

```csharp
using CerbiShield.Contracts;
using CerbiShield.Contracts.Scoring;
using Microsoft.Extensions.Logging;

namespace Cerbi.MEL.Governance;

internal class CerbiMelLogger : ILogger
{
    private readonly string _categoryName;
    private readonly CerbiMelOptions _options;
    private readonly IQueueSender _queueSender;

    public void Log<TState>(
        LogLevel logLevel,
        EventId eventId,
        TState state,
        Exception? exception,
        Func<TState, Exception?, string> formatter)
    {
        if (!IsEnabled(logLevel)) return;

        var message = formatter(state, exception);
        var logId = Guid.NewGuid().ToString();

        // Build ScoringEventDto with resolved values
        var scoringEvent = new ScoringEventDto
        {
            SchemaVersion = ContractVersions.ScoringEventSchemaVersion,
            TenantId = _options.ResolveTenantId(),
            AppName = _options.AppName ?? "unknown",
            ServiceName = _options.ServiceName,
            Environment = _options.ResolveEnvironment(),
            Runtime = "dotnet",
            LogId = logId,
            InstanceId = _options.ResolveInstanceId(),
            AppVersion = _options.ResolveAppVersion(),
            DeploymentId = _options.ResolveDeploymentId(),
            TimestampUtc = DateTime.UtcNow,
            GovernanceProfile = _options.GovernanceProfile ?? "default",
            LogLevel = MapLogLevel(logLevel),
            Score = null,  // API computes
            Violations = new List<ViolationDto>(),  // Add violations if detected
            RawPayload = new { message, categoryName = _categoryName }
        };

        var json = JsonSerializer.Serialize(scoringEvent);
        _queueSender.SendAsync(json, logId);
    }

    private static string MapLogLevel(LogLevel level) => level switch
    {
        LogLevel.Trace => "Trace",
        LogLevel.Debug => "Debug",
        LogLevel.Information => "Information",
        LogLevel.Warning => "Warning",
        LogLevel.Error => "Error",
        LogLevel.Critical => "Critical",
        _ => "Information"
    };

    // ... other ILogger members
}
```

### 5. Update Extension Method

**File:** `Cerbi.MEL.Governance/CerbiMelExtensions.cs`

```csharp
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Logging;

namespace Cerbi.MEL.Governance;

public static class CerbiMelExtensions
{
    public static ILoggingBuilder AddCerbiGovernance(
        this ILoggingBuilder builder,
        Action<CerbiMelOptions> configure)
    {
        var options = new CerbiMelOptions();
        configure(options);

        builder.Services.AddSingleton(options);
        builder.Services.AddSingleton<ILoggerProvider, CerbiMelLoggerProvider>();

        return builder;
    }
}
```

## Usage

```csharp
// Simple - TenantId from config
builder.Logging.AddCerbiGovernance(options => options
    .WithAppName("MyApp")
    .WithServiceName("payment-api")
    .WithGovernanceConfig()  // TenantId from cerbi_governance.json
    .WithQueue(connectionString, queueName)
);

// Override TenantId (for testing)
builder.Logging.AddCerbiGovernance(options => options
    .WithTenantId("test-tenant")
    .WithAppName("MyApp")
    .WithQueue(connectionString, queueName)
);
```
